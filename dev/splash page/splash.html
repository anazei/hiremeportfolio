function drawRectangle() { ticker++; //slow down rotation by if(counter % N == 0) to run on every N frames //if(ticker % 2 == 0) { //orientation of direction rotateAngleCounter = (rotateAngleCounter + ANIMATION_ROTATION_SPEED) % 360; // //clear existing bars while(spectrumContainer.numChildren) { spectrumContainer.removeChildAt(0); } for(var i=0;i<360;i+=1) { var rect = new createjs.Shape(); var height = streamData[i] / 2; if(height < 10 ) //|| i > streamData.length - 1 { height = 10; } rect.graphics.beginFill("#A4C5D9").drawRect(0, 0, 2, height, 0, 0, 0, 0); var direction = (i + rotateAngleCounter) * DEGREES_TO_RADIANS; //fixation + orientation rect.x = Math.cos(direction) * RADIUS + canvas.width/2; rect.y = Math.sin(direction) * RADIUS + canvas.height/2; rect.regX = 2.5; rect.regY = 0; rect.rotation = i - 90 + rotateAngleCounter; //90 degree offset because canvas draws 0 degrees at 90 degree point spectrumContainer.addChild(rect); //comment this line out if you only want to see the particles and not audio bars //random chance that a particle will be generated only if mod by randomNumber, lower the likelier var randomNumber = 50; //the lower this number is, the more particles will be generated var startX; var startY; if(ticker % Math.round(Math.random() * randomNumber) == 0) { //this ensures that the particles will start at the top of each bar startX = Math.round(Math.cos(direction) * (RADIUS + height) + canvas.width/2); startY = Math.round(Math.sin(direction) * (RADIUS + height) + canvas.height/2); //generate new particle on each bar redraw generateNewParticle(startX, startY, direction, height); } } } manageEmitter(); renderParticles(); //recursive function to run animation window.requestAnimationFrame(drawRectangle); }
